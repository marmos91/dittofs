# cthon04's programs for dt (test/suites.yaml), in upstream's order, each run on its own:
# upstream's runtests stop a group at its first failure, which hides every test after it.
# The words after -- are a program's command line, with the arguments upstream gives it,
# run in cthon04's tree. A line may give its own limit, as 20m; the suite gives each 10
# minutes.

# basic: test1 builds a tree that test2 removes, so the two share a directory. Each later
# test deletes its directory and makes it again, and has one of its own: a file test5
# unlinked under a write delegation stays a while as a .nfs file the client won't delete,
# and would fail the next test on test5's leftover.
basic/test1 -- env NFSTESTDIR={mount}/basic basic/test1 -t
basic/test2 -- env NFSTESTDIR={mount}/basic basic/test2 -t
basic/test3 -- env NFSTESTDIR={mount}/basic3 basic/test3 -t
basic/test4 -- env NFSTESTDIR={mount}/basic4 basic/test4 -t
basic/test5 -- env NFSTESTDIR={mount}/basic5 basic/test5 -t
basic/test6 -- env NFSTESTDIR={mount}/basic6 basic/test6 -t
basic/test7 -- env NFSTESTDIR={mount}/basic7 basic/test7 -t
basic/test8 -- env NFSTESTDIR={mount}/basic8 basic/test8 -t
basic/test9 -- env NFSTESTDIR={mount}/basic9 basic/test9 -t

# general copies its sources to the mount, and compiles and runs them there; upstream counts
# it as one test.
general/all -- env -C general NFSTESTDIR={mount}/general sh ./runtests -t

# special: each program works in its working directory, the mount's root.
special/op_unlk -- env -C {mount} TMPDIR= {tools}/cthon04/special/op_unlk
special/op_ren -- env -C {mount} TMPDIR= {tools}/cthon04/special/op_ren
special/op_chmod -- env -C {mount} TMPDIR= {tools}/cthon04/special/op_chmod
special/dupreq -- env -C {mount} {tools}/cthon04/special/dupreq 100 testfile
special/excltest -- env -C {mount} {tools}/cthon04/special/excltest
special/negseek -- env -C {mount} {tools}/cthon04/special/negseek testfile
special/rename -- env -C {mount} {tools}/cthon04/special/rename 100
special/truncate -- env -C {mount} {tools}/cthon04/special/truncate
special/holey -- env -C {mount} {tools}/cthon04/special/holey
special/nfsidem -- env -C {mount} {tools}/cthon04/special/nfsidem 50 testdir
special/rewind -- env -C {mount} {tools}/cthon04/special/rewind
special/telldir -- env -C {mount} {tools}/cthon04/special/telldir
special/freesp -- env -C {mount} {tools}/cthon04/special/freesp
special/bigfile -- env -C {mount} {tools}/cthon04/special/bigfile -s 30 bigfile
special/bigfile2 -- env -C {mount} {tools}/cthon04/special/bigfile2 bigfile2

# lock: no option set mounts with nolock, so the locks reach the server.
lock/tlocklfs -- lock/tlocklfs -r {mount}
lock/tlock64 -- lock/tlock64 -r {mount}
